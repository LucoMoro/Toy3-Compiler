def sommac(a: double; d: double; b: double; ref size: string; ref result: double)
{
    double_var = 1.00001 : double;

            switch (double_var)
                1.01 : "funziono anche sui double" !>>; stop;
                1.001 : "non dovrei essere qui" !>>; stop;
                1.00001 : "ok, sui double non funziono" !>>; stop;
            end switch;

    result := a + b + c + d;

    if (result > 100) then {
        size := "grande";
    }
    else {
        if (result > 50) then {
            size := "media";
        }
        else {
            size := "piccola";
        }
    }
}

def stampa(messaggio: string): string
{
    i: 0;
    while (i < 4) do {
        "" !>>; // Ritorno a capo
        i := i + 1;
    }
    messaggio !>>;
    return "ok";
}
